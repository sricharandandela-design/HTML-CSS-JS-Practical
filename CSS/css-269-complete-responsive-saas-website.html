/* Responsive nav: links hidden on mobile, shown from 600px up */
.cap-nav .links{ display:none; }
@media (min-width:600px){ .cap-nav .links{ display:flex; } }

/* Dashboard: flex row on desktop, stacked column on mobile */
.cap-dash{ display:flex; gap:0.8rem; }
@media (max-width:600px){ .cap-dash{ flex-direction:column; } }

/* Pricing: CSS Grid with auto-fit so columns adapt to width automatically */
.cap-pricing{ display:grid; grid-template-columns:repeat(auto-fit, minmax(110px, 1fr)); gap:0.8rem; }
