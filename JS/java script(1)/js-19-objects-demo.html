Demonstrate Objects in JavaScript

Creating an object literal, reading and updating properties, and calling a method.

const student = {
  name: "Ananya",
  rollNo: 21,
  marks: [78, 92, 85],
  average(){
    return this.marks.reduce((a, b) => a + b, 0) / this.marks.length;
  }
};

student.marks.push(90);   // update an existing property
student.grade = "A";      // add a new property

console.log(student.name, student.average(), student.grade);
