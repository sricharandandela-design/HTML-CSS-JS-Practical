Demonstrate JavaScript Functions

Function declarations, function expressions, default parameters and rest parameters, side by side.

function greet(name){ return `Hello, ${name}!`; }              // declaration
const square = function(x){ return x * x; };                    // expression
function power(base, exp = 2){ return base ** exp; }            // default parameter
function total(...nums){ return nums.reduce((a, b) => a + b, 0); } // rest parameter

console.log(greet("Student"));   // Hello, Student!
console.log(square(6));          // 36
console.log(power(5));           // 25  (exp defaults to 2)
console.log(total(1, 2, 3, 4));  // 10
