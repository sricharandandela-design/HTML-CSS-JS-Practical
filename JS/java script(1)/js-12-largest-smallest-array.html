Find the Largest and Smallest Element in an Array

Scans the array once, tracking the running maximum and minimum.

Demo

Output will appear here.

Source

function findMinMax(numbers){
  let max = numbers[0], min = numbers[0];
  for (const n of numbers){
    if (n > max) max = n;
    if (n < min) min = n;
  }
  return { max, min };
}

console.log(findMinMax([12, 45, 3, 88, 27, 6])); // { max: 88, min: 3 }
